```

│   │   │   └── design/           # Design 画布 + 节点
│   │   ├── services/              # API 客户端 + 工具函数
│   │   │   ├── clutchState.ts     # WebSocket store（全局状态）
│   │   │   ├── runApi.ts          # HTTP API 封装
│   │   │   └── terminalOrchestraUtils.ts  # Handoff/Dispatch 工具
│   │   └── types.ts               # 前端类型

│
├── services/orchestrator/         # 后端（唯一后端业务根）
│   └── src/
│       ├── main.py                # FastAPI 入口 + WS + 生命周期
│       ├── routes/                # 模块化路由 (chat/design/models/pty/...)
│       ├── compiler/              # Workflow JSON → LangGraph
│       ├── design/                # Design 模式服务
│       │   ├── session_store.py   # 会话持久化
│       │   ├── generator.py       # LLM 生成流水线
│       │   ├── layout.py          # 画布布局计算
│       │   └── preview_manager.py # Vite 预览服务器
│       ├── adapters/              # CLI 适配器
│       │   ├── claude_cli_adapter.py
│       │   ├── agy_cli_adapter.py
│       │   └── opencode_zen_adapter.py
│       ├── llm/                   # LLM Provider Router
│       │   ├── router.py
│       │   └── http_complete.py
│       ├── engine_router.py       # Agent → CLI / LLM / Hybrid 路由
│       ├── shell_session.py       # 长驻 bash PTY
│       ├── shell_exec_runtime.py  # SHELL_EXEC 执行器
│       ├── interactive_pty_runtime.py  # INTERACTIVE_PTY 执行器
│       ├── dispatch_parse.py      # 派发 @ 语法解析
│       ├── handoff_writer.py      # Handoff 文件生成
│       ├── handoff_summarizer.py  # 上下文总结
│       ├── terminal_orchestra.py  # PTY Lane 管理
│       ├── agent_storage.py       # Agent CRUD
│       ├── workflow_storage.py    # 工作流 CRUD
│       ├── state.py               # ClutchState 初始值
│       └── graph.py               # LangGraph 最小运行时
│
├── packages/shared-types/         # 前后端共享 TypeScript 类型
│   └── index.ts                   # ClutchState、WS 事件契约
│
├── workflows/                     # 内置只读模板 + JSON Schema
│   └── workflow.schema.json       # 工作流 Schema（五种节点类型）
│
├── docs/                          # 产品/架构/安装/治理文档
├── e2e/                           # Playwright 端到端测试
├── memory/                        # 跨会话六件套
├── scripts/                       # 构建/校验/发版脚本 (~30 个)
└── specs/                         # 历史需求快照
```
